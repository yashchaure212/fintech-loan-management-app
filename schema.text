///////////////////////////////////////////////////////////////////////////////
// ROLE
///////////////////////////////////////////////////////////////////////////////

model Role {
  id          String   @id @default(uuid())
  name        String   @unique
  description String?

  users User[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@map("roles")
}

///////////////////////////////////////////////////////////////////////////////
// USER
///////////////////////////////////////////////////////////////////////////////

model User {
  id String @id @default(uuid())

  roleId String
  role   Role @relation(fields: [roleId], references: [id], onDelete: Restrict)

  email    String @unique
  phone    String @unique
  password String

  isEmailVerified Boolean @default(false)
  isPhoneVerified Boolean @default(false)

  isActive  Boolean @default(true)
  isDeleted Boolean @default(false)

  deletedAt DateTime?

  /*
  |--------------------------------------------------------------------------
  | Relations
  |--------------------------------------------------------------------------
  */

  profile CustomerProfile?

  notifications Notification[] @relation("UserNotifications")

  refreshTokens RefreshToken[] @relation("UserRefreshTokens")

  auditLogs AuditLog[] @relation("UserAuditLogs")

  adminRemarks ApplicationRemark[] @relation("AdminRemarks")

  timelineEntries ApplicationTimeline[] @relation("TimelineCreatedBy")

  statusChanges LoanApplicationStatusHistory[] @relation("StatusChangedBy")

  verifiedDocuments Document[] @relation("DocumentVerifiedBy")

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@index([email])
  @@index([phone])
  @@index([roleId])

  @@map("users")
}

///////////////////////////////////////////////////////////////////////////////
// LOAN APPLICATION

model LoanApplication {
  id String @id @default(uuid())

  customerProfileId String

  customerProfile CustomerProfile @relation(
    fields: [customerProfileId],
    references: [id],
    onDelete: Restrict
  )

  loanTypeId String

  loanType LoanType @relation(
    fields: [loanTypeId],
    references: [id],
    onDelete: Restrict
  )

  applicationNumber String @unique

  status LoanApplicationStatus @default(DRAFT)

  /*
  |--------------------------------------------------------------------------
  | Loan Details
  |--------------------------------------------------------------------------
  */

  requestedAmount Decimal @db.Decimal(15,2)

  approvedAmount Decimal? @db.Decimal(15,2)

  interestRate Decimal? @db.Decimal(5,2)

  tenureMonths Int?

  processingFee Decimal? @db.Decimal(15,2)

  disbursedAmount Decimal? @db.Decimal(15,2)

  /*
  |--------------------------------------------------------------------------
  | Financial Snapshot
  |--------------------------------------------------------------------------
  */

  monthlyIncome Decimal? @db.Decimal(15,2)

  monthlyExpenses Decimal? @db.Decimal(15,2)

  existingEMIs Decimal? @db.Decimal(15,2)

  creditScore Int?

  /*
  |--------------------------------------------------------------------------
  | Employment Snapshot
  |--------------------------------------------------------------------------
  */

  employmentType EmploymentType?

  employerName String?

  occupation String?

  /*
  |--------------------------------------------------------------------------
  | Dates
  |--------------------------------------------------------------------------
  */

  submittedAt DateTime?

  approvedAt DateTime?

  rejectedAt DateTime?

  disbursedAt DateTime?

  closedAt DateTime?

  /*
  |--------------------------------------------------------------------------
  | Admin
  |--------------------------------------------------------------------------
  */

  rejectionReason String?

  adminRemarks String?

  /*
  |--------------------------------------------------------------------------
  | Relations
  |--------------------------------------------------------------------------
  */

  documents Document[]

  timelines ApplicationTimeline[]

  remarks ApplicationRemark[]

  statusHistory LoanApplicationStatusHistory[]

  emiSchedules EMISchedule[]

  createdAt DateTime @default(now())

  updatedAt DateTime @updatedAt

  @@index([customerProfileId])

  @@index([loanTypeId])

  @@index([status])

  @@index([applicationNumber])

  @@index([submittedAt])

  @@map("loan_applications")
}

model Document {
  id String @id @default(uuid())

  applicationId String

  application LoanApplication @relation(
    fields: [applicationId],
    references: [id],
    onDelete: Cascade
  )

  documentTypeId String

  documentType DocumentType @relation(
    fields: [documentTypeId],
    references: [id],
    onDelete: Restrict
  )

  fileName String

  originalFileName String

  fileUrl String

  cloudinaryPublicId String

  mimeType String

  fileSize Int

  status DocumentStatus @default(PENDING)

  adminRemark String?

  verifiedAt DateTime?

  verifiedById String?

  verifiedBy User? @relation(
    "DocumentVerifiedBy",
    fields: [verifiedById],
    references: [id],
    onDelete: SetNull
  )

  createdAt DateTime @default(now())

  updatedAt DateTime @updatedAt

  @@index([applicationId])

  @@index([documentTypeId])

  @@index([status])

  @@map("documents")
}

model EMISchedule {
  id String @id @default(uuid())

  applicationId String

  application LoanApplication @relation(
    fields: [applicationId],
    references: [id],
    onDelete: Restrict
  )

  installmentNumber Int

  dueDate DateTime

  principalAmount Decimal @db.Decimal(15,2)

  interestAmount Decimal @db.Decimal(15,2)

  emiAmount Decimal @db.Decimal(15,2)

  paidAmount Decimal @default(0) @db.Decimal(15,2)

  remainingBalance Decimal @db.Decimal(15,2)

  payments Payment[]

  createdAt DateTime @default(now())

  updatedAt DateTime @updatedAt

  @@unique([applicationId, installmentNumber])

  @@index([applicationId])

  @@index([dueDate])

  @@index([status])

  @@map("emi_schedules")
}
